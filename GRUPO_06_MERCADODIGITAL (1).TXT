#include <stdio.h>

struct telefones {
    char fixo[15];
    char movel[16];
};

struct clientes {
    int codigo;
    char nome[50];
    double cpf;
    char sexo;
    struct telefones tel;
};

struct modelos {
    char marca[20];
    char modelo[20];
};

struct produtos {
    int codigo;
    char nome[20];
    struct modelos modelo;
    double valor;
    int estoque;
};

struct carrinhos {
    struct clientes cliente;
    struct produtos produto[3];
    int quantidade[3];
    double subtotal;
    double desconto;
    double total;
};

void mostraMenuInicial();
void limpaEntrada();
void voltaMenu();
struct clientes cadastrarCliente();
int geraCodigoCliente(double cpf);
void listarClientes(struct clientes cliente[3]);
struct produtos cadastrarProduto();
int geraCodigoProduto(char nomeProduto[30]);
void listarProdutos(struct produtos produto[10]);
void efetuarVenda(struct clientes cliente[3], struct produtos produto[10], struct carrinhos carrinho);
struct clientes localizaCliente(int codigo, struct clientes cliente[3]);
struct produtos localizaProduto(int codigo, struct produtos produto[10]);
void finalizarVenda(struct carrinhos carrinho);
void listarCarrinho(struct carrinhos carrinho);
double calculaSubtotal(struct carrinhos carrinho);
double calculaTotal(double subtotal, double desconto);

void main() {
    struct clientes cliente[3];
    struct produtos produto[10];
    struct carrinhos carrinho;
    struct clientes novoCliente;
    struct produtos novoProduto;
    int i, opcao, posicao, duplicado, qtdClientes, qtdProdutos;

    for (i = 0; i < 3; i++) {
        cliente[i].codigo = 0;
    }
    for (i = 0; i < 10; i++) {
        produto[i].codigo = -1;
    }

    opcao = -1;
    while (opcao != 0) {
        mostraMenuInicial();
        opcao = -1;
        scanf("%d", &opcao);
        limpaEntrada();

        if (opcao == 1) {
            posicao = -1;
            for (i = 0; i < 3; i++) {
                if (cliente[i].codigo == 0 && posicao == -1) {
                    posicao = i;
                }
            }
            if (posicao == -1) {
                printf("\nLimite de 3 clientes cadastrados atingido!\n");
            } else {
                novoCliente = cadastrarCliente();
                duplicado = 0;
                for (i = 0; i < 3; i++) {
                    if (cliente[i].codigo == novoCliente.codigo) {
                        duplicado = 1;
                    }
                }
                if (duplicado == 1) {
                    printf("\nJa existe um cliente com o codigo %d! Cadastro cancelado.\n", novoCliente.codigo);
                } else {
                    cliente[posicao] = novoCliente;
                    printf("\nCliente cadastrado com sucesso! Codigo: %d\n", novoCliente.codigo);
                }
            }
            voltaMenu();
        } else if (opcao == 2) {
            posicao = -1;
            for (i = 0; i < 10; i++) {
                if (produto[i].codigo == -1 && posicao == -1) {
                    posicao = i;
                }
            }
            if (posicao == -1) {
                printf("\nLimite de 10 produtos cadastrados atingido!\n");
            } else {
                novoProduto = cadastrarProduto();
                duplicado = 0;
                for (i = 0; i < 10; i++) {
                    if (produto[i].codigo == novoProduto.codigo) {
                        duplicado = 1;
                    }
                }
                if (duplicado == 1) {
                    printf("\nJa existe um produto com o codigo %d! Cadastro cancelado.\n", novoProduto.codigo);
                } else {
                    produto[posicao] = novoProduto;
                    printf("\nProduto cadastrado com sucesso! Codigo: %d\n", novoProduto.codigo);
                }
            }
            voltaMenu();
        } else if (opcao == 3) {
            listarClientes(cliente);
            voltaMenu();
        } else if (opcao == 4) {
            listarProdutos(produto);
            voltaMenu();
        } else if (opcao == 5) {
            qtdClientes = 0;
            qtdProdutos = 0;
            for (i = 0; i < 3; i++) {
                if (cliente[i].codigo != 0) {
                    qtdClientes++;
                }
            }
            for (i = 0; i < 10; i++) {
                if (produto[i].codigo != -1) {
                    qtdProdutos++;
                }
            }
            if (qtdClientes == 0 || qtdProdutos == 0) {
                printf("\nCadastre pelo menos 1 cliente e 1 produto antes de efetuar uma venda.\n");
            } else {
                efetuarVenda(cliente, produto, carrinho);
            }
            voltaMenu();
        } else if (opcao == 0) {
            printf("\nSaindo do programa... Ate logo!\n");
        } else {
            printf("\nOpcao invalida! Escolha um numero de 0 a 5.\n");
            voltaMenu();
        }
    }
}

void mostraMenuInicial() {
    printf("\n========== MERCADO DIGITAL ==========\n");
    printf("1 - Cadastrar cliente\n");
    printf("2 - Cadastrar produto\n");
    printf("3 - Listar clientes\n");
    printf("4 - Listar produtos\n");
    printf("5 - Efetuar venda\n");
    printf("0 - Sair\n");
    printf("=====================================\n");
    printf("Escolha uma opcao: ");
}

void limpaEntrada() {
    char c;
    do {
        c = getchar();
    } while (c != '\n');
}

void voltaMenu() {
    printf("\nPressione ENTER para voltar ao menu...");
    limpaEntrada();
}

struct clientes cadastrarCliente() {
    struct clientes c;

    printf("\n--- Cadastro de cliente ---\n");

    printf("Nome completo: ");
    scanf(" %49[^\n]", c.nome);
    limpaEntrada();

    do {
        printf("CPF (somente numeros, sem ponto e traco): ");
        c.cpf = 0;
        scanf("%lf", &c.cpf);
        limpaEntrada();
        if (c.cpf < 100000 || c.cpf > 99999999999.0) {
            printf("CPF invalido! Digite os 11 numeros do CPF, ex: 97734545610\n");
        }
    } while (c.cpf < 100000 || c.cpf > 99999999999.0);

    do {
        printf("Sexo (M/F): ");
        scanf(" %c", &c.sexo);
        limpaEntrada();
        if (c.sexo == 'm') {
            c.sexo = 'M';
        }
        if (c.sexo == 'f') {
            c.sexo = 'F';
        }
        if (c.sexo != 'M' && c.sexo != 'F') {
            printf("Sexo invalido! Digite M ou F.\n");
        }
    } while (c.sexo != 'M' && c.sexo != 'F');

    printf("Telefone fixo (ex: (43) 3210-1000): ");
    scanf(" %14[^\n]", c.tel.fixo);
    limpaEntrada();

    printf("Celular (ex: (43) 99999-1000): ");
    scanf(" %15[^\n]", c.tel.movel);
    limpaEntrada();

    c.codigo = geraCodigoCliente(c.cpf);

    return c;
}

int geraCodigoCliente(double cpf) {
    int codigo;
    codigo = (int) (cpf / 100000);
    return codigo;
}

void listarClientes(struct clientes cliente[3]) {
    int i, achou;

    achou = 0;
    printf("\n--- Clientes cadastrados ---\n");
    for (i = 0; i < 3; i++) {
        if (cliente[i].codigo != 0) {
            printf("Codigo: %d | Nome: %s | CPF: %.0lf | Sexo: %c\n",
                   cliente[i].codigo, cliente[i].nome, cliente[i].cpf, cliente[i].sexo);
            printf("   Fixo: %s | Celular: %s\n", cliente[i].tel.fixo, cliente[i].tel.movel);
            achou = 1;
        }
    }
    if (achou == 0) {
        printf("Nenhum cliente cadastrado.\n");
    }
}

struct produtos cadastrarProduto() {
    struct produtos p;
    char nomeProduto[30];

    printf("\n--- Cadastro de produto ---\n");

    printf("Nome do produto: ");
    scanf(" %19[^\n]", p.nome);
    limpaEntrada();

    printf("Marca: ");
    scanf(" %19[^\n]", p.modelo.marca);
    limpaEntrada();

    printf("Modelo: ");
    scanf(" %19[^\n]", p.modelo.modelo);
    limpaEntrada();

    do {
        printf("Valor (maior que 0): ");
        p.valor = 0;
        scanf("%lf", &p.valor);
        limpaEntrada();
        if (p.valor <= 0) {
            printf("Valor invalido! Digite um numero maior que 0, ex: 49.90\n");
        }
    } while (p.valor <= 0);

    do {
        printf("Quantidade em estoque (maior que 0): ");
        p.estoque = 0;
        scanf("%d", &p.estoque);
        limpaEntrada();
        if (p.estoque <= 0) {
            printf("Estoque invalido! Digite um numero inteiro maior que 0.\n");
        }
    } while (p.estoque <= 0);

    nomeProduto[0] = p.modelo.marca[0];
    nomeProduto[1] = p.modelo.marca[1];
    nomeProduto[2] = p.modelo.modelo[0];
    nomeProduto[3] = p.modelo.modelo[1];
    nomeProduto[4] = '\0';
    p.codigo = geraCodigoProduto(nomeProduto);

    return p;
}

int geraCodigoProduto(char nomeProduto[30]) {
    int hash;
    hash = nomeProduto[0] * 29791 + nomeProduto[1] * 961 + nomeProduto[2] * 31 + nomeProduto[3];
    return hash;
}

void listarProdutos(struct produtos produto[10]) {
    int i, achou;

    achou = 0;
    printf("\n--- Produtos cadastrados ---\n");
    for (i = 0; i < 10; i++) {
        if (produto[i].codigo != -1) {
            printf("Codigo: %d | Nome: %s | Marca: %s | Modelo: %s | Valor: R$ %.2lf | Estoque: %d\n",
                   produto[i].codigo, produto[i].nome, produto[i].modelo.marca, produto[i].modelo.modelo, produto[i].valor, produto[i].estoque);
            achou = 1;
        }
    }
    if (achou == 0) {
        printf("Nenhum produto cadastrado.\n");
    }
}

void efetuarVenda(struct clientes cliente[3], struct produtos produto[10], struct carrinhos carrinho) {
    struct produtos prodEncontrado;
    int i, j, codCliente, codProduto, qtd, itens, continuar, jaExiste, jaNoCarrinho, disponivel;

    itens = 0;

    for (i = 0; i < 3; i++) {
        carrinho.quantidade[i] = 0;
        carrinho.produto[i].codigo = -1;
    }
    carrinho.subtotal = 0;
    carrinho.desconto = 0;
    carrinho.total = 0;

    printf("\n========== NOVA VENDA ==========\n");
    listarClientes(cliente);

    do {
        printf("\nDigite o codigo do cliente que esta comprando: ");
        codCliente = 0;
        scanf("%d", &codCliente);
        limpaEntrada();

        carrinho.cliente = localizaCliente(codCliente, cliente);

        if (carrinho.cliente.codigo == 0) {
            printf("Cliente nao encontrado! Confira o codigo na lista acima.\n");
        }
    } while (carrinho.cliente.codigo == 0);

    printf("Cliente selecionado: %s\n", carrinho.cliente.nome);

    do {
        listarProdutos(produto);

        printf("\nDigite o codigo do produto desejado: ");
        codProduto = -1;
        scanf("%d", &codProduto);
        limpaEntrada();

        prodEncontrado = localizaProduto(codProduto, produto);

        if (prodEncontrado.codigo == -1) {
            printf("Produto nao encontrado! Confira o codigo na lista acima.\n");
        } else {
            jaExiste = -1;
            for (i = 0; i < itens; i++) {
                if (carrinho.produto[i].codigo == prodEncontrado.codigo && jaExiste == -1) {
                    jaExiste = i;
                }
            }

            jaNoCarrinho = 0;
            if (jaExiste != -1) {
                jaNoCarrinho = carrinho.quantidade[jaExiste];
            }
            disponivel = prodEncontrado.estoque - jaNoCarrinho;

            if (jaExiste == -1 && itens == 3) {
                printf("O carrinho ja tem 3 tipos de produto! So e possivel aumentar a quantidade dos itens que ja estao nele.\n");
            } else if (disponivel <= 0) {
                printf("Sem estoque disponivel para %s!\n", prodEncontrado.nome);
            } else {
                do {
                    printf("Digite a quantidade desejada de %s (disponivel: %d): ", prodEncontrado.nome, disponivel);
                    qtd = 0;
                    scanf("%d", &qtd);
                    limpaEntrada();
                    if (qtd <= 0) {
                        printf("A quantidade deve ser maior que zero!\n");
                    }
                    if (qtd > disponivel) {
                        printf("Estoque insuficiente! Disponivel: %d unidade(s).\n", disponivel);
                    }
                } while (qtd <= 0 || qtd > disponivel);

                if (jaExiste != -1) {
                    carrinho.quantidade[jaExiste] = carrinho.quantidade[jaExiste] + qtd;
                    printf("Unidades somadas ao item que ja estava no carrinho! Total deste item: %d\n", carrinho.quantidade[jaExiste]);
                } else {
                    carrinho.produto[itens] = prodEncontrado;
                    carrinho.quantidade[itens] = qtd;
                    itens++;
                    printf("Produto adicionado ao carrinho! (%d de 3 tipos usados)\n", itens);
                }
            }
        }

        do {
            printf("\nDeseja adicionar outro produto? (1 - Sim / 0 - Finalizar compra): ");
            continuar = -1;
            scanf("%d", &continuar);
            limpaEntrada();
            if (continuar != 0 && continuar != 1) {
                printf("Resposta invalida! Digite 1 ou 0.\n");
            }
        } while (continuar != 0 && continuar != 1);
    } while (continuar == 1);

    if (itens > 0) {
        finalizarVenda(carrinho);

        for (i = 0; i < itens; i++) {
            for (j = 0; j < 10; j++) {
                if (produto[j].codigo == carrinho.produto[i].codigo) {
                    produto[j].estoque = produto[j].estoque - carrinho.quantidade[i];
                }
            }
        }
        printf("Estoque atualizado!\n");
    } else {
        printf("\nNenhum produto adicionado. Venda cancelada.\n");
    }
}

struct clientes localizaCliente(int codigo, struct clientes cliente[3]) {
    struct clientes encontrado;
    int i;

    encontrado.codigo = 0;
    for (i = 0; i < 3; i++) {
        if (cliente[i].codigo == codigo && codigo != 0) {
            encontrado = cliente[i];
        }
    }
    return encontrado;
}

struct produtos localizaProduto(int codigo, struct produtos produto[10]) {
    struct produtos encontrado;
    int i;

    encontrado.codigo = -1;
    for (i = 0; i < 10; i++) {
        if (produto[i].codigo == codigo && codigo != -1) {
            encontrado = produto[i];
        }
    }
    return encontrado;
}

void finalizarVenda(struct carrinhos carrinho) {
    double valorDesconto;

    listarCarrinho(carrinho);

    do {
        printf("\nDigite o percentual de desconto (ex: 10 para 10%%, 0 para sem desconto): ");
        carrinho.desconto = -1;
        scanf("%lf", &carrinho.desconto);
        limpaEntrada();
        if (carrinho.desconto < 0 || carrinho.desconto > 100) {
            printf("Percentual invalido! Informe um valor entre 0 e 100.\n");
        }
    } while (carrinho.desconto < 0 || carrinho.desconto > 100);

    carrinho.subtotal = calculaSubtotal(carrinho);
    carrinho.total = calculaTotal(carrinho.subtotal, carrinho.desconto);
    valorDesconto = carrinho.subtotal - carrinho.total;

    printf("\n========== FECHAMENTO DA VENDA ==========\n");
    printf("Subtotal:          R$ %.2lf\n", carrinho.subtotal);
    printf("Desconto (%.2lf%%): - R$ %.2lf\n", carrinho.desconto, valorDesconto);
    printf("Total a pagar:     R$ %.2lf\n", carrinho.total);
    printf("=========================================\n");
    printf("Venda finalizada com sucesso!\n");
}

void listarCarrinho(struct carrinhos carrinho) {
    int i;
    double totalItem;

    printf("\n========== RESUMO DO CARRINHO ==========\n");
    printf("Cliente: %s (Codigo: %d | CPF: %.0lf)\n", carrinho.cliente.nome, carrinho.cliente.codigo, carrinho.cliente.cpf);
    printf("----------------------------------------\n");

    for (i = 0; i < 3; i++) {
        if (carrinho.quantidade[i] > 0) {
            totalItem = carrinho.produto[i].valor * carrinho.quantidade[i];
            printf("Item %d: %s (%s %s) | Cod: %d\n", i + 1, carrinho.produto[i].nome,
                   carrinho.produto[i].modelo.marca, carrinho.produto[i].modelo.modelo, carrinho.produto[i].codigo);
            printf("   Qtd: %d x R$ %.2lf = R$ %.2lf\n", carrinho.quantidade[i], carrinho.produto[i].valor, totalItem);
        }
    }

    printf("========================================\n");
}

double calculaSubtotal(struct carrinhos carrinho) {
    double soma;
    int i;

    soma = 0;
    for (i = 0; i < 3; i++) {
        if (carrinho.quantidade[i] > 0) {
            soma = soma + carrinho.produto[i].valor * carrinho.quantidade[i];
        }
    }
    return soma;
}

double calculaTotal(double subtotal, double desconto) {
    double total;
    total = subtotal - (subtotal * (desconto / 100));
    return total;
}
